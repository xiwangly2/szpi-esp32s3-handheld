cmake_minimum_required(VERSION 3.16)
project(video_clock_tests C)
enable_testing()
add_executable(video_clock_tests test_video_clock.c)
target_include_directories(video_clock_tests PRIVATE ../../firmware/handheld/main)
if(MSVC)
    target_compile_options(video_clock_tests PRIVATE /W4 /WX /UNDEBUG)
else()
    target_compile_options(video_clock_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME video_clock COMMAND video_clock_tests)
