cmake_minimum_required(VERSION 3.16)
project(ip_camera_tests C)
enable_testing()
set(IDF_PATH "$ENV{IDF_PATH}" CACHE PATH "ESP-IDF checkout containing http_parser")
set(MAIN ${CMAKE_CURRENT_SOURCE_DIR}/../../firmware/handheld/main)
set(MULTIPART ${CMAKE_CURRENT_SOURCE_DIR}/../../firmware/handheld/components/multipart-parser)
if(NOT EXISTS "${IDF_PATH}/components/http_parser/http_parser.c")
    message(FATAL_ERROR "Set -DIDF_PATH to the pinned ESP-IDF checkout")
endif()
add_library(ipcam_parsers STATIC ${MULTIPART}/multipart_parser.c
    ${IDF_PATH}/components/http_parser/http_parser.c)
target_include_directories(ipcam_parsers PUBLIC ${MULTIPART} ${IDF_PATH}/components/http_parser)
add_executable(ip_camera_tests test_ip_camera.c ${MAIN}/ip_camera_stream.c)
target_include_directories(ip_camera_tests PRIVATE ${MAIN})
target_link_libraries(ip_camera_tests PRIVATE ipcam_parsers)
set_property(TARGET ip_camera_tests PROPERTY C_STANDARD 11)
if(MSVC)
    target_compile_options(ip_camera_tests PRIVATE /W4 /WX /UNDEBUG)
else()
    target_compile_options(ip_camera_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME ip_camera_protocol COMMAND ip_camera_tests)
set_tests_properties(ip_camera_protocol PROPERTIES TIMEOUT 30)
