cmake_minimum_required(VERSION 3.16)
project(wifi_qr_tests C)
enable_testing()
add_executable(wifi_qr_tests test_wifi_qr.c
    ../../firmware/handheld/main/wifi_qr_payload.c)
target_include_directories(wifi_qr_tests PRIVATE ../../firmware/handheld/main)
set_property(TARGET wifi_qr_tests PROPERTY C_STANDARD 11)
if(MSVC)
    target_compile_options(wifi_qr_tests PRIVATE /W4 /WX /UNDEBUG)
else()
    target_compile_options(wifi_qr_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME wifi_qr_payload COMMAND wifi_qr_tests)

set(COMPONENTS ${CMAKE_CURRENT_SOURCE_DIR}/../../firmware/handheld/managed_components)
set(QUIRC ${COMPONENTS}/espressif__quirc/quirc/lib)
set(QRGEN ${COMPONENTS}/lvgl__lvgl/src/extra/libs/qrcode)
if(EXISTS ${QUIRC}/quirc.c AND EXISTS ${QRGEN}/qrcodegen.c)
    add_library(qr_test_lib STATIC ${QUIRC}/quirc.c ${QUIRC}/decode.c
        ${QUIRC}/identify.c ${QUIRC}/version_db.c ${QRGEN}/qrcodegen.c)
    target_include_directories(qr_test_lib PUBLIC ${QUIRC} ${QRGEN})
    target_compile_definitions(qr_test_lib PRIVATE QUIRC_FLOAT_TYPE=float LV_CONF_SKIP=1)
    add_executable(wifi_qr_decode_tests test_wifi_qr_decode.c
        ../../firmware/handheld/main/wifi_qr_payload.c)
    target_include_directories(wifi_qr_decode_tests PRIVATE ../../firmware/handheld/main)
    target_link_libraries(wifi_qr_decode_tests PRIVATE qr_test_lib)
    if(MSVC)
        target_compile_options(wifi_qr_decode_tests PRIVATE /W4 /WX /UNDEBUG)
    else()
        target_compile_options(wifi_qr_decode_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        target_link_libraries(wifi_qr_decode_tests PRIVATE m)
    endif()
    add_test(NAME wifi_qr_decode COMMAND wifi_qr_decode_tests)
else()
    message(STATUS "Run idf.ps1 reconfigure to enable quirc image-decoder tests")
endif()
