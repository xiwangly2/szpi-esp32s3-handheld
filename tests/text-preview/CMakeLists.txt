cmake_minimum_required(VERSION 3.16)
project(text_preview_tests C)
enable_testing()
set(FATFS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../firmware/handheld/components/fatfs/src")
add_library(text_preview_core OBJECT ../../firmware/handheld/main/text_preview_encoding.c "${FATFS_DIR}/ffunicode.c")
target_include_directories(text_preview_core PRIVATE ../../firmware/handheld/main "${FATFS_DIR}")
set_property(TARGET text_preview_core PROPERTY C_STANDARD 11)
if(MSVC)
  target_compile_options(text_preview_core PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/fatfs_unicode_shim.h" /W4 /UNDEBUG)
else()
  target_compile_options(text_preview_core PRIVATE -include "${CMAKE_CURRENT_SOURCE_DIR}/fatfs_unicode_shim.h" -Wall -Wextra -UNDEBUG)
endif()
add_executable(text_preview_tests test_text_preview.c $<TARGET_OBJECTS:text_preview_core>)
target_include_directories(text_preview_tests PRIVATE ../../firmware/handheld/main)
set_property(TARGET text_preview_tests PROPERTY C_STANDARD 11)
if(MSVC)
  target_compile_options(text_preview_tests PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(text_preview_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME text_preview_encoding COMMAND text_preview_tests)
